import { fetchDocs } from '#/utils/graphique'
import { Exports } from '#/components/Exports'

export const getStaticProps = async () => {
  const docs = await fetchDocs()

  return {
    props: {
      ssg: {
        docs
      }
    }
  }
}

  return summary;
}
